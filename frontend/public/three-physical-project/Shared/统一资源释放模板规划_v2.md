# 统一资源释放模板规划

## 0. 设计原则

1. **幂等性**: `dispose()` 可安全重复调用
2. **彻底性**: 覆盖 Three.js + 第三方库 + DOM + 事件
3. **一致性**: 所有实验使用统一的函数签名
4. **可组合性**: 子组件可递归调用各自的 `dispose()`

---

## 1. 统一接口设计

### 1.1 函数命名
- **主入口函数**: `dispose()` - 暴露给前端统一调用
- **内部辅助函数**:
  - `trackedAddEventListener()` - 统一事件注册，自动追踪用于销毁
  - `removeTrackedEventListeners()` - 解绑所有追踪的事件
  - `disposeObject3D(obj)` - 递归清理单个 Object3D 及其 children
  - `disposeSceneGraph(root)` - 遍历清理整个场景图
  - `disposeMaterial(material)` - 深度清理材质及其关联的纹理/uniforms
  - `stopAnimationLoops()` - 停止所有动画循环和定时器

### 1.2 事件注册规范
```javascript
// ❌ 禁止：匿名函数无法被追踪和手动解绑
trackedAddEventListener(window, 'resize', () => { /* ... */ });

// ✅ 推荐：使用命名函数，便于后续维护
function onResize() { /* ... */ }
trackedAddEventListener(window, 'resize', onResize);

// ✅ 特殊场景：需要主动解绑时，保持函数引用
function disposeCustom() {
    container.removeEventListener('pointerdown', onPointerDown);
}
```
> ⚠️ 若必须使用匿名函数，销毁时只能通过 `trackedAddEventListener` 统一解绑，无法单独移除。

### 1.2 标准模板结构

```javascript
// === 统一资源释放模板 (增强版) ===
let isDisposed = false;
let animationFrameId = null;
const activeTimeoutIds = new Set();
const activeIntervalIds = new Set();

// A. 追踪函数
const activeEventListeners = [];

function trackedRequestAnimationFrame(callback) {
    const id = requestAnimationFrame((...args) => {
        if (isDisposed) return;
        callback(...args);
    });
    animationFrameId = id;
    return id;
}

function trackedSetTimeout(fn, delay) {
    const id = window.setTimeout(() => {
        activeTimeoutIds.delete(id);
        if (isDisposed) return;
        fn();
    }, delay);
    activeTimeoutIds.add(id);
    return id;
}

function trackedSetInterval(fn, delay) {
    const id = window.setInterval(() => {
        if (isDisposed) return;
        fn();
    }, delay);
    activeIntervalIds.add(id);
    return id;
}

/**
 * 统一事件注册追踪
 * 用法: trackedAddEventListener(window, 'resize', handleResize);
 *       trackedAddEventListener(container, 'pointerdown', onPointerDown);
 */
function trackedAddEventListener(target, event, handler, options) {
    target.addEventListener(event, handler, options);
    activeEventListeners.push({ target, event, handler, options });
}

function removeTrackedEventListeners() {
    for (const { target, event, handler, options } of activeEventListeners) {
        target.removeEventListener(event, handler, options);
    }
    activeEventListeners.length = 0;
}

// B. 核心清理函数
function disposeMaterial(material) {
    if (!material) return;
    const textureKeys = [
        'map', 'lightMap', 'bumpMap', 'normalMap', 'specularMap',
        'envMap', 'alphaMap', 'gradientMap', 'emissiveMap',
        'metalnessMap', 'roughnessMap', 'aoMap', 'displacementMap', 'matcap'
    ];
    for (const key of textureKeys) {
        if (material[key]) { material[key].dispose(); material[key] = null; }
    }
    // 清理 ShaderMaterial uniforms 中的纹理
    if (material.uniforms) {
        for (const uniform of Object.values(material.uniforms)) {
            if (uniform?.value instanceof THREE.Texture) { uniform.value.dispose(); }
        }
    }
    material.dispose();
}

function disposeObject3D(obj) {
    if (!obj) return;
    if (obj.children?.length > 0) {
        for (const child of obj.children) { disposeObject3D(child); }
    }
    if (obj.geometry) { obj.geometry.dispose(); obj.geometry = null; }
    if (obj.material) {
        const materials = Array.isArray(obj.material) ? obj.material : [obj.material];
        for (const mat of materials) { disposeMaterial(mat); }
        obj.material = null;
    }
}

function disposeSceneGraph(root) {
    if (!root) return;
    root.traverse((obj) => disposeObject3D(obj));
}

function stopAnimationLoops() {
    if (animationFrameId !== null) { cancelAnimationFrame(animationFrameId); animationFrameId = null; }
    for (const id of activeTimeoutIds) { clearTimeout(id); }
    activeTimeoutIds.clear();
    for (const id of activeIntervalIds) { clearInterval(id); }
    activeIntervalIds.clear();
    // 解绑所有追踪的事件
    removeTrackedEventListeners();
}

// C. 第三方库清理
function disposeThirdPartyLibraries() {
    if (typeof gsap !== 'undefined' && gsap.killTweensOf) { gsap.killTweensOf('*'); }
    if (typeof gui !== 'undefined' && gui?.destroy) { gui.destroy(); gui = null; }
}

// D. UI/DOM 清理
function disposeUI() {
    const uiContainers = ['log-container', 'ai-box', 'ai-text', 'result-list'];
    for (const id of uiContainers) {
        const el = document.getElementById(id);
        if (el) el.innerHTML = '';
    }
    // CSS2DRenderer DOM 物理移除
    if (labelRenderer?.domElement?.parentNode) {
        labelRenderer.domElement.parentNode.removeChild(labelRenderer.domElement);
    }
    // 兜底清理：类名兜底，防止重复创建导致残留
    document.querySelectorAll('.label-container, .lil-gui').forEach(el => el.remove());
}

// E. 子组件递归清理
function disposeChildComponents() {
    if (particleRenderer?.dispose) { particleRenderer.dispose(); }
    if (postProcessor?.dispose) { postProcessor.dispose(); }
}

// F. 主清理函数
function dispose() {
    if (isDisposed) return;
    isDisposed = true;
    stopAnimationLoops(); // 包含事件解绑
    disposeThirdPartyLibraries();
    disposeUI();
    disposeChildComponents();
    disposeSceneGraph(scene);
    composer?.dispose();
    controls?.dispose();
    labelRenderer?.dispose?.();

    // 渲染器彻底销毁
    if (renderer) {
        renderer.dispose();
        renderer.renderLists?.dispose();
        if (renderer.forceContextLoss) renderer.forceContextLoss();
        // 物理移除 Canvas 元素，防止 DOM 残留引用
        if (renderer.domElement?.parentNode) {
            renderer.domElement.parentNode.removeChild(renderer.domElement);
        }
    }

    // 断开全局引用，帮助垃圾回收
    scene = null;
    camera = null;
    renderer = null;
    labelRenderer = null;
    composer = null;
    controls = null;
}

// G. 注册页面卸载事件
window.addEventListener('beforeunload', dispose);
window.addEventListener('pagehide', dispose);

// 暴露给前端调用的统一接口
window.disposeExperiment = dispose;
```

> 注: 如果实验未使用某些模块（如 `labelRenderer`、`particleRenderer`），对应的清理代码会自动跳过（使用可选链 `?.`）。

## 3. 子组件 dispose() 接口规范

### 3.1 ParticleRenderer
```javascript
dispose() {
    if (this.uniforms?.uPointTexture?.value) {
        this.uniforms.uPointTexture.value.dispose();
    }
    if (this.points?.geometry) { this.points.geometry.dispose(); }
    if (this.points?.material) { this.points.material.dispose(); }
    if (this.points && this.scene) { this.scene.remove(this.points); }
}
```

### 3.2 PostProcessor
```javascript
dispose() {
    if (this.trailTimer) { clearTimeout(this.trailTimer); this.trailTimer = null; }
    this.composer?.dispose();
}
```

## 4. 实施优先级

| 优先级 | 实验 | 工作量 | 关键缺失 |
|-------|------|-------|---------|
| P0 | 玻尔理论 | 高 | 完整重写 dispose + labelRenderer |
| P0 | 电子云模型 | 高 | 完整重写 dispose + 事件追踪 |
| P1 | 光速视角 | 中 | 子组件 dispose + 主循环 + 事件追踪 |
| P1 | 霍曼转移 | 中 | raf + BufferGeometry |
| P2 | 开普勒定律 | 中 | raf + geometry dispose |
| P2 | 光电效应 | 低 | controls + beforeunload |
| P3 | Alpha/宇宙速度 | 低 | 已是范例 |

## 5. 前端统一调用方式

```javascript
// 页面切换时自动清理
window.addEventListener('beforeunload', () => {
    window.disposeExperiment?.();
});

// 前端手动切换实验时调用
function switchExperiment(experimentId) {
    window.disposeExperiment?.();
    loadExperiment(experimentId);
}
```
